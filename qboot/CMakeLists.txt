cmake_minimum_required(VERSION 3.20)
project(qboot C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Build type" FORCE)
endif()

option(Q_ENABLE_ASAN "Build with AddressSanitizer" OFF)
option(Q_ENABLE_TSAN "Build with ThreadSanitizer" OFF)
option(Q_ENABLE_UBSAN "Build with UndefinedBehaviorSanitizer" OFF)
option(Q_BUILD_HTTP  "Build libq_http (requires libevent)" ON)

set(CMAKE_C_FLAGS_DEBUG   "-g -O0")
set(CMAKE_C_FLAGS_RELEASE "-O2 -DNDEBUG")

add_compile_options(-Wall -Wextra -Wshadow -Wno-unused-parameter)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-fno-omit-frame-pointer)
endif()

if(Q_ENABLE_ASAN)
    add_compile_options(-fsanitize=address)
    add_link_options(-fsanitize=address)
endif()
if(Q_ENABLE_TSAN)
    add_compile_options(-fsanitize=thread -g)
    add_link_options(-fsanitize=thread)
endif()
if(Q_ENABLE_UBSAN)
    add_compile_options(-fsanitize=undefined)
    add_link_options(-fsanitize=undefined)
endif()

find_package(Threads REQUIRED)

add_subdirectory(core)
add_subdirectory(log)
add_subdirectory(conf)
add_subdirectory(db)
add_subdirectory(mapper)
add_subdirectory(httpc)
add_subdirectory(registry)
if(Q_BUILD_HTTP)
    add_subdirectory(http)
endif()
add_subdirectory(samples)
add_subdirectory(tools)
add_subdirectory(tests)

message(STATUS "qboot: build=${CMAKE_BUILD_TYPE} asan=${Q_ENABLE_ASAN} tsan=${Q_ENABLE_TSAN} http=${Q_BUILD_HTTP}")
